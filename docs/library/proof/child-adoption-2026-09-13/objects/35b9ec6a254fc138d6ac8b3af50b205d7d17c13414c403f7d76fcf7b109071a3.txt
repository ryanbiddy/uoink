"""Documentary text hashes only. No candidate/kernel/model execution."""
from pathlib import Path
import hashlib
import json

HERE = Path(__file__).resolve().parent
sha = lambda raw: hashlib.sha256(raw).hexdigest()
mapping = json.loads((HERE / "SOURCE-INPUTS.json").read_bytes())
for name, path in mapping["source_paths"].items():
    assert sha(Path(path).read_bytes()) == mapping["source_sha256"][name]
manifest = HERE / "PREPARATION-MANIFEST.json"
assert not manifest.exists()
rows = []
for path in sorted(HERE.rglob("*")):
    if path.is_file() and path != manifest:
        raw = path.read_bytes()
        raw.decode("utf-8-sig")
        rows.append({"path": path.relative_to(HERE).as_posix(), "bytes": len(raw), "sha256": sha(raw)})
payload = {"schema": "uoink.source-proposal-manifest.v1", "scope": "Minimal reviewed pipe selection; no candidate execution",
           "payload_count": len(rows), "payload_bytes": sum(row["bytes"] for row in rows), "files": rows}
manifest.write_text(json.dumps(payload, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"manifest": str(manifest), "payload_count": len(rows), "payload_bytes": payload["payload_bytes"], "manifest_sha256": sha(manifest.read_bytes())}, indent=2))
outline = HERE.parent / "production-connection-next01"
sources = [
    HERE.parent / "asr-production-adapter-proposal01/asr_loading_adapter.py",
    HERE.parent / "asr-production-adapter-proposal01/PORT-CONTRACTS.md",
    HERE.parent / "asr-worker-namespace-proposal01/snapshot_lifecycle.py",
    HERE.parent / "asr-worker-namespace-proposal01/owned_generation_protocol.py",
    HERE.parent / "asr-worker-namespace-proposal01/pinned_buffer_namespace.py",
    HERE.parent / "vad-owned-factory-port-proposal01/model_binding_registry.py",
    HERE.parent / "vad-owned-factory-port-proposal01/context/owned_guard.py",
    HERE.parent / "vad-owned-factory-port-proposal01/WORKER-CAPABILITY-CONTRACT.md",
    HERE / "generated_worker_flow.py",
]
bindings = []
for path in sources:
    raw = path.read_bytes()
    bindings.append({"path": str(path), "bytes": len(raw), "sha256": sha(raw)})
(outline / "SOURCE-BINDINGS.json").write_text(json.dumps({"source_only": True, "files": bindings,
    "note": "Existing migration drafts consulted for scope only; their historical counts are not current status."}, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"next_work": str(outline / "NEXT-WORK.md"), "next_work_sha256": sha((outline / "NEXT-WORK.md").read_bytes()),
    "source_bindings_sha256": sha((outline / "SOURCE-BINDINGS.json").read_bytes())}, indent=2))
