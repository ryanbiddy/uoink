"""Data-only path/hash selection revision. Does not execute candidate code."""
from pathlib import Path
import ast
import difflib
import hashlib
import json

HERE = Path(__file__).resolve().parent
PRIOR = HERE.parent / "child-readset-adoption-proposal01"
PIPE = HERE.parent / "asr-dummy-worker-timeout-proposal02" / "win32_private_pipe.py"
sha = lambda raw: hashlib.sha256(raw).hexdigest()
seal_raw = (PRIOR / "PREPARATION-MANIFEST.json").read_bytes()
assert sha(seal_raw) == "4389b252a8ef40f9fa80421f586aabd9c5273ded8e383b6a077e72a89ff775a0"
seal = json.loads(seal_raw)
assert seal["payload_count"] == 30
for row in seal["files"]:
    raw = (PRIOR / row["path"]).read_bytes()
    assert len(raw) == row["bytes"] and sha(raw) == row["sha256"]
pipe_raw = PIPE.read_bytes()
assert sha(pipe_raw) == "73a1109a55f2bc807594655c71b24a3e35eb88d7f744497df2cc328dc224cae7"
ast.parse(pipe_raw, filename=str(PIPE))
(HERE / "PRIOR-PREPARATION-MANIFEST.json").write_bytes(seal_raw)
(HERE / ".gitattributes").write_bytes((PRIOR / ".gitattributes").read_bytes())
copied = []
for name in ("owned_generation_protocol.py", "inherited_readset.py", "generated_worker_flow.py", "dummy_bootstrap.py"):
    raw = (PRIOR / name).read_bytes()
    assert not (HERE / name).exists()
    (HERE / name).write_bytes(raw)
    copied.append({"name": name, "before_sha256": sha(raw), "after_sha256": sha((HERE / name).read_bytes())})
before_launcher = (PRIOR / "run_adoption01.ps1").read_text(encoding="utf-8")
after_launcher = before_launcher.replace(str(PRIOR), str(HERE))
assert before_launcher.count(str(PRIOR)) == 1
needle = '    if($taskName -in $taskSharedNames){$taskPath=Join-Path $taskShared $taskName}\n'
assert after_launcher.count(needle) == 1
after_launcher = after_launcher.replace(needle, needle + "    if($taskName -ceq 'win32_private_pipe.py'){$taskPath='" + str(PIPE) + "'}\n")
(HERE / "run_adoption01.ps1").write_bytes(after_launcher.encode())
(HERE / "launcher-selection.diff").write_text("".join(difflib.unified_diff(before_launcher.splitlines(True), after_launcher.splitlines(True),
    fromfile="proposal01/run_adoption01.ps1", tofile="proposal02/run_adoption01.ps1")), encoding="utf-8", newline="\n")
prior_map_raw = (PRIOR / "SOURCE-INPUTS.json").read_bytes()
mapping = json.loads(prior_map_raw)
old_pipe = {"path": mapping["source_paths"]["win32_private_pipe.py"], "sha256": mapping["source_sha256"]["win32_private_pipe.py"]}
mapping["source_paths"]["win32_private_pipe.py"] = str(PIPE)
mapping["source_sha256"]["win32_private_pipe.py"] = sha(pipe_raw)
for row in copied:
    mapping["source_paths"][row["name"]] = str(HERE / row["name"])
for name, path in mapping["source_paths"].items():
    raw = Path(path).read_bytes()
    assert sha(raw) == mapping["source_sha256"][name]
    ast.parse(raw, filename=name)
new_map_raw = (json.dumps(mapping, indent=2) + "\n").encode()
(HERE / "SOURCE-INPUTS.json").write_bytes(new_map_raw)
(HERE / "source-map-selection.diff").write_text("".join(difflib.unified_diff(prior_map_raw.decode().splitlines(True), new_map_raw.decode().splitlines(True),
    fromfile="proposal01/SOURCE-INPUTS.json", tofile="proposal02/SOURCE-INPUTS.json")), encoding="utf-8", newline="\n")
for case in ("positive", "wrong_identity"):
    admission = json.loads((PRIOR / ("ROOT-ADMISSION-" + case + ".template.json")).read_bytes())
    assert admission["root_reviewed"] is False
    admission["source_inputs_sha256"] = sha(new_map_raw)
    admission["launcher_sha256"] = sha(after_launcher.encode())
    (HERE / ("ROOT-ADMISSION-" + case + ".template.json")).write_text(json.dumps(admission, indent=2) + "\n", encoding="utf-8")
protocol = (PRIOR / "PROTOCOL.md").read_text(encoding="utf-8")
protocol = protocol.replace(str(PRIOR), str(HERE))
old_paragraph = "Timeout01's ERROR_BROKEN_PIPE failure is already separate evidence. This proposal presently pins original pipe4a0cccc8, and does not silently consume the unqualified73a1109a repair."
assert protocol.count(old_paragraph) == 1
protocol = protocol.replace(old_paragraph, "Timeout01's ERROR_BROKEN_PIPE failure remains separate evidence. Revision02 explicitly selects repaired pipe73a1109a under root's generated-scope acceptance recorded in REVISION-BRIEF.md; no child-adoption execution has occurred.")
(HERE / "PROTOCOL.md").write_text(protocol, encoding="utf-8", newline="\n")
report = {"source_only": True, "candidate_executions": 0, "prior_seal_verified_payloads": 30,
          "copied_candidate_sources": copied, "old_pipe": old_pipe,
          "selected_pipe": {"path": str(PIPE), "sha256": sha(pipe_raw)},
          "source_inputs_sha256": sha(new_map_raw), "launcher_sha256": sha(after_launcher.encode()),
          "parsed_sources": 8, "changed_behavior_assertions": 0}
(HERE / "REVISION-BINDINGS.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
print(json.dumps(report, indent=2))
