"""Data-only proposed controls and source inventory. No candidate execution."""
from pathlib import Path
import ast
import difflib
import hashlib
import json

HERE = Path(__file__).resolve().parent
sha = lambda raw: hashlib.sha256(raw).hexdigest()
mapping_raw = (HERE / "SOURCE-INPUTS.json").read_bytes()
mapping = json.loads(mapping_raw)
for name, source_path in mapping["source_paths"].items():
    raw = Path(source_path).read_bytes()
    assert sha(raw) == mapping["source_sha256"][name]
    ast.parse(raw, filename=name)
bootstrap = ast.parse((HERE / "dummy_bootstrap.py").read_bytes())
symbols_node = next(node.value for node in bootstrap.body if isinstance(node, ast.Assign)
                    and any(isinstance(target, ast.Name) and target.id == "SYMBOLS" for target in node.targets))
symbols = ast.literal_eval(symbols_node)
assert len(symbols) == 33 and "SetFilePointerEx" in symbols and "GetLastError" in symbols
assert len(symbols - {"GetLastError"}) == 32
launcher = (HERE / "run_adoption01.ps1").read_bytes()
old_launcher = (HERE / "before-receipt-review01" / "run_adoption01.ps1").read_text()
(HERE / "run_adoption01.ps1.receipt-review01.diff").write_text("".join(difflib.unified_diff(old_launcher.splitlines(True), launcher.decode().splitlines(True),
    fromfile="before/run_adoption01.ps1", tofile="after/run_adoption01.ps1")), encoding="utf-8", newline="\n")
for case, directory in (("positive", "child-readset-positive01"), ("wrong_identity", "child-readset-wrong-identity01")):
    admission = {"root_reviewed": False, "scope": "five-generated-inherited-files-only", "case": case,
        "run_path": str(HERE.parent / directory), "source_inputs_sha256": sha(mapping_raw), "launcher_sha256": sha(launcher),
        "expected_controller_native_exit": 0, "expected_child_native_exit": 0 if case == "positive" else 2,
        "authority": "Template only. Root must review exact sources and record this case's admission separately."}
    (HERE / ("ROOT-ADMISSION-" + case + ".template.json")).write_text(json.dumps(admission, indent=2) + "\n", encoding="utf-8")
report = {"schema": "uoink.child-readset-static-preparation.v1", "candidate_executions": 0,
    "native_measurements": 0, "parsed_sources": 8, "fixed_dispatch_names": sorted(symbols - {"GetLastError"}),
    "dispatch_count": 32, "source_inputs_sha256": sha(mapping_raw), "launcher_sha256": sha(launcher),
    "sdk_reference": str(HERE.parent / "asr-worker-namespace-proposal01" / "SDK-PIPE-EXCERPTS.txt"),
    "sdk_reference_sha256": sha((HERE.parent / "asr-worker-namespace-proposal01" / "SDK-PIPE-EXCERPTS.txt").read_bytes()),
    "source_sha256": mapping["source_sha256"],
    "scope": "Sources parsed and bytes bound only; no model, file fixture, kernel or worker execution."}
(HERE / "STATIC-BINDINGS.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
print(json.dumps(report, indent=2))
