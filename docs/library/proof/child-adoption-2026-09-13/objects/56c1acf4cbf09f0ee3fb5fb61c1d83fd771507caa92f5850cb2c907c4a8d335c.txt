"""Generated-only inherited ReadSet adoption; no I/O or native import at import.

The fixed trusted bootstrap supplies both primitives and the authenticated
channel. Public records and wire integers are not independent authority.
"""
import hashlib
import json
from types import MappingProxyType
from pinned_buffer_namespace import BufferAsset, PinnedBufferNamespace
from win32_worker_connection import FileIdentity, ReadSet, _absolute_local_path

NAMES = ("config.json", "model.bin", "preprocessor_config.json", "tokenizer.json", "vocabulary.json")
GENERATED = MappingProxyType({name: ("Uoink generated adoption fixture: " + name + ". No model data.\n").encode("ascii")
                              for name in NAMES})
PAYLOAD_BYTES = sum(len(value) for value in GENERATED.values())
FORMAT = "uoink.generated-inherited-readset.v1"
CASES = frozenset(("positive", "wrong_identity"))


class AdoptionRefusal(RuntimeError):
    pass


def _require(condition, reason):
    if not condition:
        raise AdoptionRefusal(reason)


def _canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True,
                      allow_nan=False).encode("ascii")


def namespace_digest():
    rows = [[name, len(GENERATED[name]), hashlib.sha256(GENERATED[name]).hexdigest()] for name in NAMES]
    return hashlib.sha256(json.dumps(rows, separators=(",", ":")).encode("ascii")).hexdigest()


def manifest_digest(payload):
    return hashlib.sha256(_canonical(payload)).hexdigest()


def fixed_paths(root):
    _absolute_local_path(root)  # Lexical only; no resolve/stat/open.
    _require(not root.endswith("\\"), "fixed_generated_directory_required")
    return tuple(_absolute_local_path(root + "\\" + name) for name in NAMES)


def _identity_row(identity):
    _require(type(identity) is FileIdentity, "exact_file_identity_type")
    return {"final_path": identity.final_path, "volume_serial": identity.volume_serial,
            "file_id": identity.file_id.hex(), "size": identity.size,
            "links": identity.links, "directory": identity.directory}


def controller_manifest(primitives, read_set, worker, root, case):
    """Called serially before child resume by the actual lifecycle start port."""
    paths = fixed_paths(root)
    _require(case in CASES and type(case) is str, "fixed_generated_case")
    _require(any(item is read_set for item in primitives.read_sets)
             and any(item is worker for item in primitives.workers)
             and worker.read_set is read_set and worker.assigned and not worker.resumed
             and not worker.unconfirmed and not read_set.unconfirmed and not read_set.released,
             "exact_suspended_owned_worker_required")
    _require(len(read_set.members) == len(read_set.identities) == len(worker.child_read_handles) == 5
             and all(handle.closed for handle in worker.inherited_copies), "exact_inheritance_membership")
    rows = []
    for index, name in enumerate(NAMES):
        handle, expected = read_set.members[index], read_set.identities[index]
        primitives._owned(handle)
        _require(primitives.identity(handle) == expected, "parent_guard_identity_changed")
        row = {"name": name, "handle": worker.child_read_handles[index],
               "identity": _identity_row(expected), "sha256": hashlib.sha256(GENERATED[name]).hexdigest()}
        rows.append(row)
    if case == "wrong_identity":
        # The complete authenticated negative manifest deliberately binds the
        # wrong first file ID. No message tampering or invalid MAC is simulated.
        altered = bytearray.fromhex(rows[0]["identity"]["file_id"])
        altered[0] ^= 1
        rows[0]["identity"]["file_id"] = bytes(altered).hex()
    payload = {"schema": FORMAT, "case": case, "namespace_sha256": namespace_digest(), "members": rows}
    validate_manifest(payload, root, case, worker.child_control_handle, manifest_digest(payload))
    return payload


def validate_manifest(payload, root, case, control_value, expected_manifest_sha256):
    """Validate the entire authenticated passive message before handle calls."""
    paths = fixed_paths(root)
    _require(type(payload) is dict and set(payload) == {"schema", "case", "namespace_sha256", "members"},
             "adoption_manifest_shape")
    _require(type(case) is str and case in CASES and payload["schema"] == FORMAT
             and payload["case"] == case and payload["namespace_sha256"] == namespace_digest(),
             "fixed_adoption_scope")
    rows = payload["members"]
    _require(type(rows) is list and len(rows) == 5, "five_ordered_generated_members")
    _require(type(control_value) is int and 0 < control_value < (1 << 64) - 1, "fixed_control_handle")
    values, validated = set(), []
    for index, row in enumerate(rows):
        _require(type(row) is dict and set(row) == {"name", "handle", "identity", "sha256"}, "member_shape")
        name, value, identity = row["name"], row["handle"], row["identity"]
        _require(type(name) is str and name == NAMES[index], "member_order_and_name")
        _require(type(value) is int and 0 < value < (1 << 64) - 1 and value != control_value
                 and value not in values, "unique_inherited_file_handle")
        values.add(value)
        _require(type(row["sha256"]) is str and row["sha256"] == hashlib.sha256(GENERATED[name]).hexdigest(),
                 "generated_member_digest")
        _require(type(identity) is dict and set(identity) == {"final_path", "volume_serial", "file_id", "size", "links", "directory"},
                 "complete_file_identity")
        final_path = identity["final_path"]
        # GetFinalPathNameByHandleW(flags=0) returns this fixed extended DOS
        # spelling. Compare lexically; never follow a path supplied by a frame.
        _require(type(final_path) is str and len(final_path) <= 4096
                 and final_path.casefold() == ("\\\\?\\" + paths[index]).casefold(), "fixed_generated_final_path")
        _require(type(identity["volume_serial"]) is int and 0 <= identity["volume_serial"] < 1 << 64,
                 "volume_serial_bound")
        file_id = identity["file_id"]
        _require(type(file_id) is str and len(file_id) == 32 and all(char in "0123456789abcdef" for char in file_id),
                 "file_id_shape")
        _require(type(identity["size"]) is int and identity["size"] == len(GENERATED[name])
                 and type(identity["links"]) is int and identity["links"] == 1
                 and identity["directory"] is False, "generated_file_shape")
        expected = FileIdentity(final_path, identity["volume_serial"], bytes.fromhex(file_id),
                                identity["size"], 1, False)
        validated.append((name, value, expected, row["sha256"]))
    _require(type(expected_manifest_sha256) is str and len(expected_manifest_sha256) == 64
             and manifest_digest(payload) == expected_manifest_sha256, "authenticated_manifest_binding")
    return tuple(validated)


class InheritedReadSetAdoption:
    """One worker-local operation, owned by the fixed serial child flow.

    Refused owners are kept until that child exits. No arbitrary recovery or
    real namespace capability can be obtained by toggling a field here.
    """

    def __init__(self, primitives):
        self.primitives = primitives
        self.read_set = None
        self.assets = None
        self.namespace = None
        self.inheritance_cleared = 0
        self.identities_checked = 0
        self.materialization_started = False
        self.status = "new"

    def adopt(self, authenticated_payload, root, case, control_value, manifest_sha256):
        _require(self.status == "new" and self.read_set is None, "single_adoption_attempt")
        self.status = "validating"
        try:
            rows = validate_manifest(authenticated_payload, root, case, control_value, manifest_sha256)
            _require(not any(not handle.closed and handle.value in {row[1] for row in rows}
                             for handle in self.primitives.handles), "inherited_handle_already_owned")
            read_set = ReadSet()
            self.read_set = read_set
            self.primitives.read_sets.append(read_set)
            # All partial owners precede inheritance and identity queries. The
            # control endpoint is already separately owned, and cannot overlap.
            for name, value, expected, digest in rows:
                read_set.members.append(self.primitives._retain(value, "adopted-generated-member"))
            ffi = self.primitives.api.ffi
            for handle in read_set.members:
                flags = ffi.c_uint32()
                self.primitives.api.checked("GetHandleInformation", self.primitives._owned(handle), ffi.byref(flags))
                _require(flags.value == 1, "exact_inherited_handle_flag_required")
                self.primitives.api.checked("SetHandleInformation", self.primitives._owned(handle), 1, 0)
                self.primitives.api.checked("GetHandleInformation", self.primitives._owned(handle), ffi.byref(flags))
                _require(flags.value == 0, "inherited_flag_not_cleared")
                self.inheritance_cleared += 1
            for handle, row in zip(read_set.members, rows):
                actual = self.primitives.identity(handle)
                self.identities_checked += 1
                _require(actual == row[2], "inherited_file_identity_mismatch")
                read_set.identities.append(actual)
            self.assets = tuple(BufferAsset(row[0], row[2].size, row[3], handle)
                                for handle, row in zip(read_set.members, rows))
            self.status = "adopted"
            return self
        except BaseException:
            self.status = "refused"
            if self.read_set is not None:
                self.read_set.unconfirmed = True
            raise

    def materialize_generated(self):
        _require(self.status == "adopted" and not self.materialization_started, "adoption_before_readback")
        self.materialization_started = True
        try:
            self.namespace = PinnedBufferNamespace(self.primitives, self.read_set, self.assets, PAYLOAD_BYTES).materialize()
            _require(self.namespace.namespace_sha256 == namespace_digest(), "complete_generated_namespace_digest")
            for name in NAMES:
                _require(self.namespace.lookup(name) == GENERATED[name], "generated_complete_byte_readback")
            self.status = "materialized"
            return {"namespace_sha256": namespace_digest(), "payload_bytes": PAYLOAD_BYTES,
                    "members": [{"name": name, "bytes": len(GENERATED[name]),
                                 "sha256": hashlib.sha256(GENERATED[name]).hexdigest()} for name in NAMES]}
        except BaseException:
            self.status = "refused"
            self.read_set.unconfirmed = True
            raise

    def release_after_reads(self):
        _require(self.status == "materialized" and self.read_set is not None and not self.read_set.unconfirmed,
                 "only_successful_readback_can_release")
        # No worker belongs to this child-local set and all synchronous reads
        # have returned. The parent guards remain held through actual exit.
        result = self.primitives.release_read_set(self.read_set)
        _require(result is True and self.read_set.released, "child_read_guard_release_unconfirmed")
        self.status = "released"
        return True

    def receipt(self):
        return {"status": self.status, "owned_members": len(self.read_set.members) if self.read_set else 0,
                "inheritance_cleared": self.inheritance_cleared, "identities_checked": self.identities_checked,
                "materialization_started": self.materialization_started,
                "read_set_unconfirmed": self.read_set.unconfirmed if self.read_set else None,
                "read_set_released": self.read_set.released if self.read_set else False,
                "handles_closed": [handle.closed for handle in self.read_set.members] if self.read_set else [],
                "complete_native_namespace_protection": False, "model_calls": 0}
