$ErrorActionPreference = 'Stop'
$proposal = $PSScriptRoot
$runDir = Join-Path $proposal 'bridge-preflight01'
$output = Join-Path $runDir 'AUTHOR-INPUT-CASE-VERIFICATION.json'
if (Test-Path -LiteralPath $output) { throw 'Fresh documentary verification output required' }
function Hash-File([string]$path) { (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant() }
$pinsPath = Join-Path $proposal 'PINS.json'
$pinsHash = Hash-File $pinsPath
if ($pinsHash -cne '2e126d468dc48da025894a1f90a40a7476eaa039496861c3e1cec37087123e07') { throw 'Executed pins changed' }
$pins = Get-Content -LiteralPath $pinsPath -Raw | ConvertFrom-Json
$checks = @()
foreach ($item in $pins.files) {
    $path = Join-Path $proposal $item.path
    $hash = Hash-File $path
    if ($hash -cne $item.sha256 -or (Get-Item -LiteralPath $path).Length -ne $item.bytes) { throw 'Original pinned input changed' }
    $checks += @{kind='original';path=$item.path;before_sha256=$item.sha256;after_sha256=$hash;bytes=$item.bytes;match=$true}
}
$plan = Get-Content -LiteralPath (Join-Path $runDir 'plan.json') -Raw | ConvertFrom-Json
foreach ($item in $plan.inputs) {
    $path = Join-Path $runDir $item.name
    $hash = Hash-File $path
    if ($hash -cne $item.sha256 -or (Hash-File $item.source) -cne $item.sha256) { throw 'Copied executed input changed' }
    $checks += @{kind='executed_copy';path=$item.name;before_sha256=$item.sha256;after_sha256=$hash;bytes=(Get-Item -LiteralPath $path).Length;match=$true}
}
$report = Get-Content -LiteralPath (Join-Path $runDir 'stdout.json') -Raw | ConvertFrom-Json
$expected = Get-Content -LiteralPath (Join-Path $runDir 'EXPECTED-CASES.json') -Raw | ConvertFrom-Json
$native = Get-Content -LiteralPath (Join-Path $runDir 'NATIVE-RETURN.json') -Raw | ConvertFrom-Json
$exitRecord = Get-Content -LiteralPath (Join-Path $runDir 'actual-exit.json') -Raw | ConvertFrom-Json
$outer = Get-Content -LiteralPath (Join-Path $runDir 'outer-tool-result.json') -Raw | ConvertFrom-Json
if ($report.passed -ne 61 -or $report.failed -ne 0 -or $report.skipped -ne 0 -or $report.qualification_exit -ne 0 -or $native.native_exit -ne 0 -or $exitRecord.native_exit -ne 0 -or $outer.exit_code -ne 0) { throw 'Actual qualification result mismatch' }
if (@($report.cases).Count -ne 61 -or @($report.cases.name|Sort-Object -Unique).Count -ne 61 -or @($report.cases|Where-Object status -ne 'passed').Count -ne 0 -or ((@($report.cases.name)|ConvertTo-Json -Compress) -cne (@($expected.cases)|ConvertTo-Json -Compress))) { throw 'Case membership mismatch' }
$guardNames = @('metadata_wrappers_intact','realpath_wrapper_intact','content_access_closed','reader_real_profile_closed','bridge_real_profile_absent')
foreach ($name in $guardNames) { if ($report.guard_fields.$name -ne $true) { throw 'Guard integrity mismatch' } }
if ($report.guard_integrity -ne $true -or $report.startup_binding -ne $true -or @($report.audit_denials).Count -ne 0 -or (Get-Item -LiteralPath (Join-Path $runDir 'stderr.log')).Length -ne 0) { throw 'Guard or stream outcome mismatch' }
$pinnedCopy = Join-Path $runDir 'PINS.executed-copy.json'
if (Test-Path -LiteralPath $pinnedCopy) { throw 'Executed pin copy already exists' }
Copy-Item -LiteralPath $pinsPath -Destination $pinnedCopy
