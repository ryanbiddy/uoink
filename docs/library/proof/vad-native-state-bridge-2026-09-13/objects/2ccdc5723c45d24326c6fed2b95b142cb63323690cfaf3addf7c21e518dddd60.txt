$ErrorActionPreference = 'Stop'
$proposal = $PSScriptRoot
$runDir = Join-Path $proposal 'bridge-preflight01'
$pinsPath = Join-Path $proposal 'PINS.json'
$admissionPath = Join-Path $proposal 'ROOT-ADMISSION.json'
function Hash-File([string]$path) { (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant() }
if (-not (Test-Path -LiteralPath $admissionPath -PathType Leaf)) { throw 'Root exact-source synthetic admission required' }
$pinsHash = Hash-File $pinsPath
$admission = Get-Content -LiteralPath $admissionPath -Raw | ConvertFrom-Json
if ($admission.scope -cne 'generated-bytes-and-fake-ports-only' -or $admission.label -cne 'bridge-preflight01' -or $admission.pins_sha256 -cne $pinsHash -or $admission.admitted -ne $true) { throw 'Synthetic admission does not match frozen pins' }
$pins = Get-Content -LiteralPath $pinsPath -Raw | ConvertFrom-Json
$names = @('BRIEF.md','PORT-CONTRACT.md','PREFLIGHT01-PROTOCOL.md','SOURCE-BINDINGS.json',
    'state_bridge.py','qualify_bridge.py','EXPECTED-CASES.json','run_preflight01.ps1',
    'context/plain_state_reader.py','context/fixed-plan.json','context/fixed-factory.proposal.txt')
if ((@($pins.files.path | Sort-Object) | ConvertTo-Json -Compress) -cne (@($names | Sort-Object) | ConvertTo-Json -Compress)) { throw 'Frozen input membership mismatch' }
foreach ($item in $pins.files) {
    $path = Join-Path $proposal $item.path
    if ((Hash-File $path) -cne $item.sha256 -or (Get-Item -LiteralPath $path).Length -ne $item.bytes) { throw 'Frozen input bytes changed' }
}
if (Test-Path -LiteralPath $runDir) { throw 'Fresh synthetic run path required' }
New-Item -ItemType Directory -Path $runDir | Out-Null
Copy-Item -LiteralPath $admissionPath -Destination (Join-Path $runDir 'root-admission.json')
$childMap = @(
    @('context/plain_state_reader.py','plain_state_reader.py'),
    @('state_bridge.py','state_bridge.py'),
    @('qualify_bridge.py','qualify_bridge.py'),
    @('context/fixed-plan.json','fixed-plan.json'),
    @('EXPECTED-CASES.json','EXPECTED-CASES.json')
)
$childFiles = [ordered]@{}
$copiedInputs = @()
foreach ($pair in $childMap) {
    $original = Join-Path $proposal $pair[0]
    $copy = Join-Path $runDir $pair[1]
    Copy-Item -LiteralPath $original -Destination $copy
    $hash = Hash-File $original
    if ((Hash-File $copy) -cne $hash) { throw 'Child input copy mismatch' }
    $childFiles[$pair[1]] = $hash
    $copiedInputs += [ordered]@{name=$pair[1];source=$original;sha256=$hash}
}
[ordered]@{scope='synthetic-fake-ports-only';child_files=$childFiles;pins_sha256=$pinsHash;native_approved=$false;release_approved=$false} | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $runDir 'INPUTS.json') -Encoding utf8
$env:IG_FORBIDDEN_LIVE = 'C:\Users\hello\AppData\Local\Uoink\index.db'
Get-ChildItem Env: | Where-Object {
    $_.Name -match '(?i)(ANTHROPIC|OPENAI|GEMINI|GOOGLE_API|GROK|XAI|HF_TOKEN|HUGGING_FACE|HUGGINGFACE|API_KEY|ACCESS_TOKEN|AUTH_TOKEN|GITHUB_TOKEN|GH_TOKEN|PROXY|SSL_CERT|REQUESTS_CA_BUNDLE|CURL_CA_BUNDLE|SSLKEYLOGFILE)'
} | ForEach-Object { Remove-Item -LiteralPath ('Env:\' + $_.Name) }
$arguments = @('-I','-S','-B',(Join-Path $runDir 'qualify_bridge.py'))
$plan = [ordered]@{label='bridge-preflight01';started_utc=[DateTime]::UtcNow.ToString('o');pins_sha256=$pinsHash;inputs=$copiedInputs;arguments=$arguments;startup_binding_set=$true;environment_scrub='provider credentials/tokens/proxies/certificate overrides/key log; values omitted';scope='generated bytes and trusted fake ports only';native_or_factory_execution_approved=$false}
$plan | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $runDir 'plan.json') -Encoding utf8
& 'C:\Python314\python.exe' @arguments 1> (Join-Path $runDir 'stdout.json') 2> (Join-Path $runDir 'stderr.log')
$nativeExit = $global:LASTEXITCODE
[ordered]@{returned_utc=[DateTime]::UtcNow.ToString('o');native_exit=$nativeExit;exit_source='global:LASTEXITCODE immediately after child; never reset';postchecks_started=$false} | ConvertTo-Json -Depth 5 | Set-Content -LiteralPath (Join-Path $runDir 'NATIVE-RETURN.json') -Encoding utf8
if ($null -eq $nativeExit) { throw 'Native child exit unavailable; preserved as instrumentation failure' }
$unchanged = $true
foreach ($item in $pins.files) { if ((Hash-File (Join-Path $proposal $item.path)) -cne $item.sha256) { $unchanged = $false } }
foreach ($item in $copiedInputs) { if ((Hash-File (Join-Path $runDir $item.name)) -cne $item.sha256) { $unchanged = $false } }
$captureBound = (Get-Item -LiteralPath (Join-Path $runDir 'stdout.json')).Length -le 65536 -and (Get-Item -LiteralPath (Join-Path $runDir 'stderr.log')).Length -le 131072
$exitReceipt = [ordered]@{finished_utc=[DateTime]::UtcNow.ToString('o');native_exit=$nativeExit;inputs_unchanged=$unchanged;capture_size_bound_met=$captureBound;startup_binding_set=($env:IG_FORBIDDEN_LIVE -ceq 'C:\Users\hello\AppData\Local\Uoink\index.db');native_or_factory_execution_approved=$false}
$exitReceipt | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $runDir 'actual-exit.json') -Encoding utf8
if ($nativeExit -ne 0) { exit $nativeExit }
if (-not $unchanged -or -not $captureBound) { throw 'Synthetic input or capture integrity failure' }
$report = Get-Content -LiteralPath (Join-Path $runDir 'stdout.json') -Raw | ConvertFrom-Json
$expected = Get-Content -LiteralPath (Join-Path $runDir 'EXPECTED-CASES.json') -Raw | ConvertFrom-Json
$complete = $report.qualification_exit -eq 0 -and $report.passed -eq $expected.case_count -and $report.failed -eq 0 -and $report.skipped -eq 0 -and $report.startup_binding -eq $true -and @($report.audit_denials).Count -eq 0 -and (Get-Item -LiteralPath (Join-Path $runDir 'stderr.log')).Length -eq 0
$complete = $complete -and $report.guard_integrity -eq $true -and $report.guard_fields.metadata_wrappers_intact -eq $true -and $report.guard_fields.realpath_wrapper_intact -eq $true -and $report.guard_fields.content_access_closed -eq $true -and $report.guard_fields.reader_real_profile_closed -eq $true -and $report.guard_fields.bridge_real_profile_absent -eq $true
$complete = $complete -and ((@($report.cases.name) | ConvertTo-Json -Compress) -ceq (@($expected.cases) | ConvertTo-Json -Compress)) -and @($report.cases | Where-Object status -ne 'passed').Count -eq 0
$complete = $complete -and $report.bridge_sha256 -ceq $childFiles['state_bridge.py'] -and $report.reader_sha256 -ceq $childFiles['plain_state_reader.py'] -and $report.harness_sha256 -ceq $childFiles['qualify_bridge.py']
[ordered]@{complete=$complete;passed=$report.passed;failed=$report.failed;qualification_exit=$report.qualification_exit;native_exit=$nativeExit;elapsed_seconds=$report.elapsed_seconds;stdout_sha256=(Hash-File (Join-Path $runDir 'stdout.json'));inputs_unchanged=$unchanged;native_qualified=$false;release_approved=$false} | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $runDir 'checked-result.json') -Encoding utf8
if (-not $complete) { throw 'Synthetic qualification is incomplete or failed' }
exit 0
