def deny(event):
    EVENTS.append(event)

def allowed_path(path):
    return (isinstance(path, (str, bytes, os.PathLike))
        and ntpath.normcase(ntpath.abspath(os.fsdecode(path))) in READS)


def audit(event, args):
    if event == 'open':
        path, mode, flags = args
        if (not READS_OPEN or not allowed_path(path)
            or flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)):
            deny(event)
    elif event == 'import' and args[0] not in ALLOWED_IMPORTS:
        deny('import:' + str(args[0]))
    elif event.startswith(('socket.', 'subprocess.', 'ctypes.', 'winreg.')) or event in {
        'os.system', 'os.startfile', 'os.startfile/2', 'os.spawn', 'os.fork', 'os.exec',
        'os.add_dll_directory', 'os.listdir', 'os.scandir', 'os.mkdir', 'os.remove',
        'os.rmdir', 'os.rename', 'os.link', 'os.symlink', 'os.chmod', 'os.utime'}:
        deny(event)

def encoded(document, data=DATA):
    header = json.dumps(document, ensure_ascii=True, sort_keys=True,
        separators=(',', ':'), allow_nan=False).encode('utf-8')
    header += b' ' * (-len(header) % 8)
    return struct.pack('<Q', len(header)) + header + data

@contextmanager
def patched(owner, name, value):
    previous = getattr(owner, name)
    setattr(owner, name, value)
    try:
        yield
    finally:
        setattr(owner, name, previous)


class FakeTensor:
    def __init__(self, index, shape, address):
        self.index, self.shape, self.address = index, shape, address
        self.data = bytearray(math.prod(shape) * 4)
        self.released = False


class FakeTensorPort:
    def __init__(self):
        self.tensors = []
        self.release_attempts = []
        self.copy_arguments = []
        self.copy_count = self.read_count = self.allocate_count = 0
        self.address = 0x100000
        self.gap = 64
        self.allocate_hook = self.inspect_hook = self.copy_hook = self.read_hook = self.release_hook = None
        self.events = []

    def allocate_owned_cpu_f32(self, shape):
        self.allocate_count += 1
        if self.allocate_hook is not None:
            special = self.allocate_hook(self, shape)
            if special is not None:
                return special
        tensor = FakeTensor(len(self.tensors), shape, self.address)
        self.address += len(tensor.data) + self.gap
        self.tensors.append(tensor)
        self.events.append(('allocate', tensor.index))
        return tensor

    def inspect_owned(self, tensor):
        assert not tensor.released
        facts = bridge.TensorFacts(tensor.shape, 'F32', 'cpu', 'strided', True,
            True, False, 0, tensor.address, len(tensor.data), True, True)
        if self.inspect_hook is not None:
            return self.inspect_hook(self, tensor, facts)
        return facts

    def copy_owned_f32le(self, tensor, offset, chunk):
        assert not tensor.released and type(chunk) is bytes
        assert 0 < len(chunk) <= 65536 and len(chunk) % 4 == offset % 4 == 0
        assert 0 <= offset <= len(tensor.data) - len(chunk)
        self.copy_count += 1
        self.copy_arguments.append((type(chunk).__name__, len(chunk), chunk is BASE))
        tensor.data[offset:offset+len(chunk)] = chunk
        if self.copy_hook is not None:
            self.copy_hook(self, tensor, offset, chunk)

    def read_owned_f32le(self, tensor, offset, size):
        assert not tensor.released and size <= 65536
        self.read_count += 1
        value = bytes(tensor.data[offset:offset+size])
        if self.read_hook is not None:
            return self.read_hook(self, tensor, offset, size, value)
        return value

    def release_owned(self, tensor):
        self.release_attempts.append(tensor.index)
        self.events.append(('release', tensor.index))
        if self.release_hook is not None:
            self.release_hook(self, tensor)
        if not tensor.released:
            tensor.data[:] = b'\0' * len(tensor.data)
            tensor.released = True


class FakeFactoryPort:
    def __init__(self, tensor_port):
        self.tensor_port = tensor_port
        self.calls = self.release_attempts = self.internal_releases = 0
        self.products = []
        self.build_hook = self.release_hook = None
        self.received_keys = None

    def build_strict_owned(self, state):
        self.calls += 1
        assert type(state) is dict and all(type(key) is str for key in state)
        assert tuple(state) == tuple(key for key, _ in SCHEMA)
        assert len(state) == 54 and all(not value.released for value in state.values())
        self.received_keys = tuple(state)
        self.tensor_port.events.append(('factory', 54))
        product = {'state': {key: bytes(tensor.data) for key, tensor in state.items()}, 'released': False}
        self.products.append(product)
        try:
            if self.build_hook is not None:
                return self.build_hook(self, state, product)
            return product
        except BaseException:
            # The factory port owns partial creation before returning its result.
            product['state'].clear()
            product['released'] = True
            self.internal_releases += 1
            raise

    def release_product(self, product):
        self.release_attempts += 1
        if self.release_hook is not None:
            self.release_hook(self, product)
        product['state'].clear()
        product['released'] = True


def run(port=None, factory=None, raw=BASE, profile=None, **kwargs):
    port = FakeTensorPort() if port is None else port
    factory = FakeFactoryPort(port) if factory is None else factory
    result = bridge.exercise_synthetic_bridge(raw, approval(raw) if profile is None else profile,
        port, factory, **kwargs)
    return result, port, factory


def refused(call, phase=None, cleanup=0):
    try:
        call()
    except bridge.BridgeRefusal as error:
        if phase is not None:
            assert error.phase == phase, (phase, error.phase)
        assert error.cleanup_failures == cleanup
        assert len(str(error)) <= 150
        return error
    raise AssertionError('Expected bridge refusal')


def all_released(port):
    assert all(tensor.released for tensor in port.tensors)
    assert len(port.release_attempts) == len(set(port.release_attempts))


def before_ports(raw=BASE, profile=None):
    port = FakeTensorPort()
    factory = FakeFactoryPort(port)
    selected = approval(raw) if profile is None else profile
    refused(lambda: bridge.exercise_synthetic_bridge(raw, selected, port, factory), 'reader_validation')
    assert port.allocate_count == 0 and factory.calls == 0 and not port.release_attempts
