from __future__ import annotations

import base64
import csv
from datetime import datetime, timezone

UPSTREAM_VERSION = "1.2.1"
OUTPUT_VERSION = "1.2.1+uoink.localassets1"
EXPECTED_WHEEL_NAME = "faster_whisper-1.2.1-py3-none-any.whl"
EXPECTED_WHEEL_SIZE = 1118909
EXPECTED_WHEEL_SHA256 = "79a66ad50688c0b794dd501dc340a736992a6342f7f95e5811be60b5224a26a7"
OUTPUT_WHEEL_NAME = "faster_whisper-1.2.1+uoink.localassets1-py3-none-any.whl"
INPUT_SOURCE_SHA256 = "5d5ffb00018561d3d529b2c72e1d9f5fff055bea725f3cccc7c6c67f5cc8ffe4"
INPUT_VERSION_SHA256 = "70ee45276562706d9787f105fb3010417fa9621fe6a9e2fad01a209f3044d9c5"
ZIP_TIME = (2026, 9, 13, 0, 0, 0)
MAX_MEMBER_SIZE = 2 * 1024 * 1024
MAX_TOTAL_SIZE = 4 * 1024 * 1024
MAX_ARCHIVE_SIZE = 5 * 1024 * 1024
RECIPE_HASHES = {
    "B2.py.txt": "bf452635becacf6bba46825be6d6eea533da472ce966ee47f5ce7bccfc3bf06d",
    "NOTICE.txt": "f82a61fe4fa1dd443f24898c268282f96d1fb8a559c542ea1ab438ca2e34397e",
    "member-manifest.json": "2f1816cf0f521339d8b6709bfa27b0b6a7762582a61939f3ac901d65cdb26296",
    "patch.txt": "57451db2279d34c1e5c60f9a9d2c59b3b4a2a921607b53103533bb19a5d8e970",
}
PACKAGE_MEMBERS = frozenset("faster_whisper/" + name for name in (
    "__init__.py", "audio.py", "feature_extractor.py", "tokenizer.py", "transcribe.py",
    "utils.py", "vad.py", "version.py", "assets/__init__.py", "assets/silero_vad_v6.onnx"))

def digest(raw: bytes) -> str:
    return hashlib.sha256(raw).hexdigest()

def record_hash(raw: bytes) -> str:
    return "sha256=" + base64.urlsafe_b64encode(hashlib.sha256(raw).digest()).decode().rstrip("=")

def dist_info(version: str) -> str:
    if version not in (UPSTREAM_VERSION, OUTPUT_VERSION):
        raise ValueError("Unexpected distribution version")
    return "faster_whisper-" + version + ".dist-info"

def member_names(version: str) -> frozenset[str]:
    names = {dist_info(version) + "/" + name for name in ("LICENSE", "METADATA", "WHEEL", "top_level.txt", "RECORD")}
    if version == OUTPUT_VERSION:
        names.add(dist_info(version) + "/UOINK-LOCALASSETS-NOTICE.txt")
    return PACKAGE_MEMBERS | names

def check_name(name: str) -> None:
    if not name or "\x00" in name or "\\" in name or name.startswith("/"):
        raise ValueError("Unsafe ZIP member name")
    for part in name.split("/"):
        if part in ("", ".", "..") or ":" in part or part.rstrip(" .") != part:
            raise ValueError("Unsafe ZIP member component")
        if re.fullmatch(r"(?i)(CON|PRN|AUX|NUL|COM[0-9]|LPT[0-9])(?:\..*)?", part):
            raise ValueError("Device ZIP member")

def check_no_links(path: Path) -> None:
    for part in (*reversed(path.parents), path):
        try:
            info = part.lstat()
        except FileNotFoundError:
            continue
        if stat.S_ISLNK(info.st_mode) or getattr(info, "st_reparse_tag", 0) or getattr(info, "st_file_attributes", 0) & 0x400:
            raise ValueError("Linked or reparse path refused")

def safe_path(value: str | Path) -> Path:
    text = str(value)
    win = PureWindowsPath(text)
    if not text.strip() or "\x00" in text or text.startswith(("\\\\", "//")) or win.drive and not win.is_absolute():
        raise ValueError("Local non-device path required")
    for part in win.parts:
        if part != win.anchor:
            check_name(part)
    path = Path(os.path.abspath(text))
    check_no_links(path)
    return path

def fresh_destination(value: str | Path, wheel: Path) -> Path:
    path = safe_path(value)
    if path == wheel or path.is_relative_to(wheel) or wheel.is_relative_to(path):
        raise ValueError("Input/output overlap refused")
    if path.exists():
        raise FileExistsError("Destination reuse refused")
    if not path.parent.is_dir():
        raise FileNotFoundError("Destination parent must already exist")
    return path

def parse_metadata(raw: bytes, version: str) -> None:
    message = BytesParser().parsebytes(raw)
    expected = {"Metadata-Version": "2.1", "Name": "faster-whisper", "Version": version,
                "Requires-Python": ">=3.9", "License": "MIT", "License-File": "LICENSE"}
    if message.defects:
        raise ValueError("Malformed METADATA")
    for key, value in expected.items():
        if message.get_all(key) != [value]:
            raise ValueError("Unexpected or duplicate METADATA " + key)

def parse_wheel_metadata(raw: bytes, version: str) -> None:
    message = BytesParser().parsebytes(raw)
    generator = "bdist_wheel (0.45.1)" if version == UPSTREAM_VERSION else "uoink-localassets-wheel (1)"
    expected = {"Wheel-Version": "1.0", "Generator": generator, "Root-Is-Purelib": "true", "Tag": "py3-none-any"}
    if message.defects:
        raise ValueError("Malformed WHEEL")
    for key, value in expected.items():
        if message.get_all(key) != [value]:
            raise ValueError("Unexpected or duplicate WHEEL " + key)

def validate_archive(raw: bytes, version: str = UPSTREAM_VERSION) -> dict[str, bytes]:
    """Structural parser. Production build additionally enforces exact wheel SHA."""
    if len(raw) > MAX_ARCHIVE_SIZE:
        raise ValueError("Archive exceeds bound")
    expected = member_names(version)
    with zipfile.ZipFile(io.BytesIO(raw)) as archive:
        infos = archive.infolist()
        if len(infos) != len(expected) or archive.comment:
            raise ValueError("Unexpected ZIP member count or comment")
        seen, total = set(), 0
        for info in infos:
            check_name(info.filename)
            key = info.filename.casefold()
            if key in seen:
                raise ValueError("Duplicate/case-colliding ZIP member")
            seen.add(key)
            kind = stat.S_IFMT(info.external_attr >> 16)
            if kind not in (0, stat.S_IFREG) or info.is_dir() or info.external_attr & 0x400:
                raise ValueError("Non-regular ZIP member")
            if info.flag_bits & 1 or info.compress_type not in (zipfile.ZIP_STORED, zipfile.ZIP_DEFLATED):
                raise ValueError("Encrypted or unsupported compression")
            if info.extra or info.comment or not 0 <= info.file_size <= MAX_MEMBER_SIZE:
                raise ValueError("ZIP extras/comments/size refused")
            total += info.file_size
        if total > MAX_TOTAL_SIZE or {info.filename for info in infos} != expected:
            raise ValueError("Unexpected ZIP member set or total size")
        members = {info.filename: archive.read(info) for info in infos}
    record_name = dist_info(version) + "/RECORD"
    rows = list(csv.reader(io.StringIO(members[record_name].decode("utf-8")), strict=True))
    seen_rows = set()
    for row in rows:
        if len(row) != 3:
            raise ValueError("Malformed RECORD row")
        name, recorded, size = row
        check_name(name)
        if name in seen_rows or name not in members:
            raise ValueError("Duplicate or extraneous RECORD row")
        seen_rows.add(name)
        if name == record_name:
            if recorded or size:
                raise ValueError("Nonempty RECORD self row")
        elif recorded != record_hash(members[name]) or size != str(len(members[name])):
            raise ValueError("RECORD hash/size mismatch")
    if seen_rows != set(members):
        raise ValueError("Incomplete RECORD")
    parse_metadata(members[dist_info(version) + "/METADATA"], version)
    parse_wheel_metadata(members[dist_info(version) + "/WHEEL"], version)
    return members

def load_recipe() -> dict[str, bytes]:
    directory = safe_path(Path(__file__).absolute().parent / "recipe")
    recipe = {}
    if {path.name for path in directory.iterdir()} != set(RECIPE_HASHES):
        raise ValueError("Unexpected recipe member set")
    for name, expected in RECIPE_HASHES.items():
        path = safe_path(directory / name)
        raw = path.read_bytes()
        if digest(raw) != expected:
            raise ValueError("Recipe hash mismatch: " + name)
        recipe[name] = raw
    return recipe

def make_record(members: dict[str, bytes], version: str) -> bytes:
    record_name = dist_info(version) + "/RECORD"
    stream = io.StringIO(newline="")
    writer = csv.writer(stream, lineterminator="\n")
    for name in sorted(set(members) - {record_name}):
        writer.writerow([name, record_hash(members[name]), len(members[name])])
    writer.writerow([record_name, "", ""])
    return stream.getvalue().encode()

def transform_members(members: dict[str, bytes], recipe: dict[str, bytes]) -> dict[str, bytes]:
    """Transform inert bytes; caller must validate the archive and recipe first."""
    if set(members) != member_names(UPSTREAM_VERSION):
        raise ValueError("Unexpected source member set")
    if digest(members["faster_whisper/transcribe.py"]) != INPUT_SOURCE_SHA256 or digest(members["faster_whisper/version.py"]) != INPUT_VERSION_SHA256:
        raise ValueError("Source identity mismatch")
    if set(recipe) != set(RECIPE_HASHES) or any(digest(recipe[key]) != value for key, value in RECIPE_HASHES.items()):
        raise ValueError("Recipe identity mismatch")
    old, new = dist_info(UPSTREAM_VERSION), dist_info(OUTPUT_VERSION)
    result = {name.replace(old + "/", new + "/", 1): raw for name, raw in members.items() if name != old + "/RECORD"}
    result["faster_whisper/transcribe.py"] = recipe["B2.py.txt"]
    result["faster_whisper/version.py"] = members["faster_whisper/version.py"].replace(b'__version__ = "1.2.1"', b'__version__ = "1.2.1+uoink.localassets1"')
    metadata = result[new + "/METADATA"]
    if metadata.count(b"Version: 1.2.1\n") != 1:
        raise ValueError("Unexpected Version byte representation")
    result[new + "/METADATA"] = metadata.replace(b"Version: 1.2.1\n", b"Version: 1.2.1+uoink.localassets1\n")
    wheel = result[new + "/WHEEL"]
    if wheel.count(b"Generator: bdist_wheel (0.45.1)\n") != 1:
        raise ValueError("Unexpected Generator byte representation")
    result[new + "/WHEEL"] = wheel.replace(b"Generator: bdist_wheel (0.45.1)\n", b"Generator: uoink-localassets-wheel (1)\n")
    result[new + "/UOINK-LOCALASSETS-NOTICE.txt"] = recipe["NOTICE.txt"]
    result[new + "/RECORD"] = make_record(result, OUTPUT_VERSION)
    return result

def verify_manifest(members: dict[str, bytes], expected_rows: list[dict]) -> None:
    if len(expected_rows) != len({row["member"] for row in expected_rows}) or set(members) != {row["member"] for row in expected_rows}:
        raise ValueError("Output manifest member mismatch")
    for row in expected_rows:
        raw = members[row["member"]]
        if len(raw) != row["bytes"] or digest(raw) != row["sha256"]:
            raise ValueError("Output manifest hash/size mismatch: " + row["member"])

def serialize_members(members: dict[str, bytes]) -> bytes:
    record_name = dist_info(OUTPUT_VERSION) + "/RECORD"
    if set(members) != member_names(OUTPUT_VERSION):
        raise ValueError("Unexpected output member set")
    stream = io.BytesIO()
    with zipfile.ZipFile(stream, "w", compression=zipfile.ZIP_STORED) as archive:
        for name in sorted(set(members) - {record_name}) + [record_name]:
            info = zipfile.ZipInfo(name, ZIP_TIME)
            info.create_system = 3
            info.external_attr = 0o100644 << 16
            info.compress_type = zipfile.ZIP_STORED
            archive.writestr(info, members[name])
    return stream.getvalue()

def build(wheel_path: str | Path, output_dir: str | Path) -> dict:
    """No hash override, fixture bypass, resolver or model/package import."""
    wheel = safe_path(wheel_path)
    destination = fresh_destination(output_dir, wheel)
    if wheel.name != EXPECTED_WHEEL_NAME:
        raise ValueError("Unexpected upstream filename")
    if wheel.stat().st_size != EXPECTED_WHEEL_SIZE:
        raise ValueError("Upstream wheel size mismatch")
    raw = wheel.read_bytes()
    if len(raw) != EXPECTED_WHEEL_SIZE or digest(raw) != EXPECTED_WHEEL_SHA256:
        raise ValueError("Upstream wheel hash mismatch")
    recipe = load_recipe()
    original = validate_archive(raw)
    output = transform_members(original, recipe)
    manifest = json.loads(recipe["member-manifest.json"])
    verify_manifest(output, manifest["members"])
    blob = serialize_members(output)
    verify_manifest(validate_archive(blob, OUTPUT_VERSION), manifest["members"])
    check_no_links(destination)
    destination.mkdir(exist_ok=False)
    target = destination / OUTPUT_WHEEL_NAME
    try:
        check_no_links(target)
        with target.open("xb") as handle:
            handle.write(blob)
        check_no_links(target)
        if target.read_bytes() != blob:
            raise ValueError("Written wheel bytes drifted")
        receipt = {"status": "BUILT_AND_BYTE_VERIFIED", "release_ready": False,
            "created_utc": datetime.now(timezone.utc).isoformat(), "python": sys.version,
            "builder_sha256": digest(Path(__file__).read_bytes()), "recipe_hashes": RECIPE_HASHES,
            "upstream": {"path": str(wheel), "filename": EXPECTED_WHEEL_NAME, "size": len(raw), "sha256": digest(raw)},
            "output": {"path": str(target), "filename": OUTPUT_WHEEL_NAME, "version": OUTPUT_VERSION, "size": len(blob), "sha256": digest(blob)},
            "input_members": {name: {"bytes": len(data), "sha256": digest(data)} for name, data in sorted(original.items())},
            "output_members": manifest["members"], "zip": manifest["zip"],
            "dependencies_imported": False, "model_execution": False}
        with (destination / "provenance.json").open("x", encoding="utf-8", newline="\n") as handle:
            handle.write(json.dumps(receipt, indent=2) + "\n")
        return receipt
    except Exception as exc:
        check_no_links(destination)
        with (destination / "failure.json").open("x", encoding="utf-8", newline="\n") as handle:
            handle.write(json.dumps({"status": "FAILED", "error_type": type(exc).__name__, "error": str(exc), "release_ready": False}) + "\n")
        raise

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--wheel", required=True)
    parser.add_argument("--out-dir", required=True)
    args = parser.parse_args()
    print(json.dumps(build(args.wheel, args.out_dir)))
