$ErrorActionPreference = 'Stop'
$scratch = 'E:\AI\projects\uoink\checkouts\Yoink-library\_scratch'
$author = Join-Path $scratch 'vad-owned-cpu-tensor-port-proposal01'
$destination = Join-Path $scratch 'astra-owned-cpu-tensor-port-qualification01'
$expectedPins = '7b68ee7c7c4febf6a8e089443358688e2ed40786c01663b573c17426dfd727f9'
function Hash-File([string]$path) { (Get-FileHash -LiteralPath $path -Algorithm SHA256).Hash.ToLowerInvariant() }
$pinsPath = Join-Path $author 'PINS.json'
if ((Hash-File $pinsPath) -cne $expectedPins) { throw 'Author pins differ' }
$pins = Get-Content -LiteralPath $pinsPath -Raw | ConvertFrom-Json
if (@($pins.files).Count -ne 13) { throw 'Expected exactly thirteen author inputs' }
if (Test-Path -LiteralPath $destination) { throw 'Independent root destination must be fresh' }
foreach ($entry in $pins.files) {
    if ($entry.path -match '(^/|\\|:|(^|/)\.\.(/|$))') { throw 'Unexpected pinned path' }
    $source = Join-Path $author $entry.path
    if ((Hash-File $source) -cne $entry.sha256 -or (Get-Item -LiteralPath $source).Length -ne $entry.bytes) { throw 'Author source differs' }
}
New-Item -ItemType Directory -Path $destination | Out-Null
$bindings = @()
foreach ($entry in $pins.files) {
    $source = Join-Path $author $entry.path
    $target = Join-Path $destination $entry.path
    $parentDirectory = Split-Path -Parent $target
    if (-not (Test-Path -LiteralPath $parentDirectory)) { New-Item -ItemType Directory -Path $parentDirectory | Out-Null }
    Copy-Item -LiteralPath $source -Destination $target
    $copied = Hash-File $target
    if ($copied -cne $entry.sha256 -or (Hash-File $source) -cne $entry.sha256) { throw 'Independent source copy differs' }
    $bindings += [ordered]@{path=$entry.path;source=$source;destination=$target;bytes=$entry.bytes;sha256=$copied}
}
Copy-Item -LiteralPath $pinsPath -Destination (Join-Path $destination 'PINS.json')
if ((Hash-File (Join-Path $destination 'PINS.json')) -cne $expectedPins) { throw 'Copied pin manifest differs' }
[ordered]@{prepared_utc=[DateTime]::UtcNow.ToString('o');pins_sha256=$expectedPins;copied_input_count=13;inputs=$bindings;execution_admission_created=$false;qualification_executed=$false;native_approved=$false} | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath (Join-Path $destination 'COPY-BINDINGS.json') -Encoding utf8
Write-Output 'Thirteen exact source inputs and original PINS copied. No admission or execution created.'
