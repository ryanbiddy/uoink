"""Parser negatives for the local wheel builder; no model imports or execution."""
import base64
import csv
import hashlib
import importlib.util
import io
import sys
import zipfile
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[1]
spec = importlib.util.spec_from_file_location('wheel_boundary_builder', ROOT / 'scripts/build_nltk_pathsec_wheel.py')
builder = importlib.util.module_from_spec(spec)
spec.loader.exec_module(builder)


def fixture(kind):
    dist = 'nltk-3.10.3.dist-info/'
    data = {'nltk/VERSION': b'3.10.3\n',
            dist+'METADATA': b'Metadata-Version: 2.4\nName: nltk\nVersion: 3.10.3\n',
            dist+'WHEEL': b'Wheel-Version: 1.0\nRoot-Is-Purelib: true\nTag: py3-none-any\n'}
    if kind == 'metadata': data[dist+'METADATA'] += b'Name: other\n'
    if kind == 'case': data['NLTK/VERSION'] = b'3.10.3\n'
    if kind == 'backslash': data['nltk/part\\NUL'] = b'file'
    if kind == 'wheel_tag': data[dist+'WHEEL'] += b'Tag: cp313-cp313-manylinux_x86_64\n'
    output = io.StringIO()
    writer = csv.writer(output, lineterminator='\n')
    rows = [[name, 'sha256='+base64.urlsafe_b64encode(hashlib.sha256(raw).digest()).rstrip(b'=').decode(), str(len(raw))]
            for name, raw in data.items()]
    writer.writerows(rows)
    if kind == 'record': writer.writerow(rows[0])
    if kind != 'self_missing': writer.writerow([dist+'RECORD', '', ''])
    data[dist+'RECORD'] = output.getvalue().encode()
    archive = io.BytesIO()
    with zipfile.ZipFile(archive, 'w') as stream:
        for name, raw in data.items(): stream.writestr(name, raw)
    archive.seek(0)
    return archive


@pytest.mark.parametrize('kind', ['metadata', 'case', 'backslash', 'record', 'wheel_tag', 'self_missing'])
def test_ambiguous_archives_are_rejected(kind):
    with zipfile.ZipFile(fixture(kind)) as archive:
        with pytest.raises(ValueError): builder.validate_zip_archive_members(archive)


def test_valid_synthetic_archive_is_readable():
    with zipfile.ZipFile(fixture('valid')) as archive:
        rows, _, version = builder.validate_zip_archive_members(archive)
    assert version == '3.10.3' and len(rows) == 3


def test_filesystem_probe_failure_is_not_swallowed(monkeypatch):
    def denied(*args, **kwargs): raise PermissionError('probe denied')
    monkeypatch.setattr(Path, 'lstat', denied)
    with pytest.raises(PermissionError, match='probe denied'):
        builder.safe_local_path('fixture.whl')


def test_unrelated_caller_import_does_not_change_validation(tmp_path, monkeypatch):
    monkeypatch.setitem(sys.modules, 'nltk', object())
    wheel = tmp_path / builder.EXPECTED_WHEEL_NAME
    wheel.write_bytes(b'short')
    with pytest.raises(ValueError, match='size mismatch'):
        builder.build_nltk_wheel(wheel, tmp_path / 'result')
