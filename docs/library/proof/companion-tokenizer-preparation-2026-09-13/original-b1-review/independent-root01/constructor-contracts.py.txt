"""Six reviewed B assertions; loaded only by the guarded stdlib harness."""
from __future__ import annotations
import ast
import copy
import os
from pathlib import Path
import tempfile
from types import SimpleNamespace
import unittest

def blocked(*args, **kwargs):
    raise AssertionError("unexpected runtime or download call")


def whisper_constructor_prefix(source):
    """Keep real constructor control flow through tokenizer assignment only.

    Later feature extraction is outside this contract; stopping before it is
    explicit and is not a claim to execute the complete model constructor.
    """
    tree = ast.parse(Path(source).read_text(encoding="utf-8-sig"))
    cls = next(node for node in tree.body if isinstance(node, ast.ClassDef) and node.name == "WhisperModel")
    init = copy.deepcopy(next(node for node in cls.body if isinstance(node, ast.FunctionDef) and node.name == "__init__"))
    stop = next(index for index, node in enumerate(init.body) if any(
        isinstance(child, ast.Attribute) and isinstance(child.value, ast.Name)
        and child.value.id == "self" and child.attr == "hf_tokenizer"
        and isinstance(child.ctx, ast.Store) for child in ast.walk(node)))
    init.body = init.body[:stop + 1]
    events = []
    token = object()

    def from_file(path):
        events.append("tokenizer.file")
        if Path(path).read_bytes() == b"invalid":
            raise ValueError("synthetic invalid tokenizer")
        return token

    def from_buffer(data):
        events.append("tokenizer.bytes")
        return token

    def from_pretrained(name):
        events.append("tokenizer.remote")
        return token

    def constructor(*args, **kwargs):
        events.append("constructor")
        return SimpleNamespace(is_multilingual=True)

    scope = {
        "os": os, "get_logger": lambda: None, "download_model": blocked,
        "ctranslate2": SimpleNamespace(models=SimpleNamespace(Whisper=constructor)),
        "tokenizers": SimpleNamespace(Tokenizer=SimpleNamespace(
            from_file=from_file, from_buffer=from_buffer, from_pretrained=from_pretrained)),
    }
    future = ast.ImportFrom(module="__future__", names=[ast.alias(name="annotations")], level=0)
    module = ast.fix_missing_locations(ast.Module(body=[future, init], type_ignores=[]))
    exec(compile(module, str(source), "exec"), scope)
    return scope, events, token


class CompanionConstructorContracts(unittest.TestCase):
    def setUp(self):
        self.directory = tempfile.TemporaryDirectory(prefix="uoink-tokenizer-synthetic-")
        self.addCleanup(self.directory.cleanup)
        self.root = Path(self.directory.name)
        self.scope, self.events, self.token = whisper_constructor_prefix(FW_SOURCE)

    def test_missing_local_tokenizer_refuses_before_constructor(self):
        with self.assertRaises(FileNotFoundError):
            self.scope["__init__"](SimpleNamespace(), str(self.root), local_files_only=True)
        self.assertEqual(self.events, [])

    def test_invalid_local_tokenizer_refuses_before_constructor(self):
        (self.root / "tokenizer.json").write_bytes(b"invalid")
        with self.assertRaises(ValueError):
            self.scope["__init__"](SimpleNamespace(), str(self.root), local_files_only=True)
        self.assertEqual(self.events, ["tokenizer.file"])

    def test_valid_local_tokenizer_is_prepared_before_constructor(self):
        (self.root / "tokenizer.json").write_bytes(b"synthetic tokenizer")
        model = SimpleNamespace()
        self.scope["__init__"](model, str(self.root), local_files_only=True)
        self.assertEqual(self.events, ["tokenizer.file", "constructor"])
        self.assertIs(model.hf_tokenizer, self.token)

    def test_tokenizer_disappearing_after_file_check_still_refuses_before_constructor(self):
        tokenizer = self.root / "tokenizer.json"
        tokenizer.write_bytes(b"synthetic tokenizer")
        def disappear(path):
            self.events.append("tokenizer.file")
            tokenizer.unlink()
            raise FileNotFoundError("synthetic disappearance after positive file check")
        self.scope["tokenizers"].Tokenizer.from_file = disappear
        with self.assertRaises(FileNotFoundError):
            self.scope["__init__"](SimpleNamespace(), str(self.root), local_files_only=True)
        self.assertEqual(self.events, ["tokenizer.file"])

    def test_explicit_nonlocal_fallback_is_preserved(self):
        model = SimpleNamespace()
        self.scope["__init__"](model, str(self.root), local_files_only=False)
        self.assertEqual(self.events, ["constructor", "tokenizer.remote"])
        self.assertIs(model.hf_tokenizer, self.token)

    def test_supplied_tokenizer_bytes_precede_constructor(self):
        model = SimpleNamespace()
        self.scope["__init__"](model, "synthetic-memory-model", local_files_only=True,
                               files={"tokenizer.json": b"synthetic tokenizer", "model.bin": b"not a model"})
        self.assertEqual(self.events, ["tokenizer.bytes", "constructor"])
