+class EmptyBufferContracts(unittest.TestCase):
+    def setUp(self):
+        self.directory = tempfile.TemporaryDirectory(prefix="uoink-empty-buffer-synthetic-")
+        self.addCleanup(self.directory.cleanup)
+        self.root = Path(self.directory.name)
+        self.scope, self.events, self.token = whisper_constructor_prefix(FW_SOURCE)
+
+    def _check_empty_buffer(self, local_only, ambient_file):
+        if ambient_file:
+            (self.root / "tokenizer.json").write_bytes(b"ambient synthetic tokenizer")
+
+        def reject_empty_buffer(data):
+            self.events.append("tokenizer.bytes")
+            if data == b"":
+                raise ValueError("synthetic empty tokenizer buffer")
+            return self.token
+
+        self.scope["tokenizers"].Tokenizer.from_buffer = reject_empty_buffer
+        model = SimpleNamespace()
+        error = None
+        try:
+            self.scope["__init__"](
+                model, str(self.root), local_files_only=local_only,
+                files={"tokenizer.json": b"", "model.bin": b"not a model"},
+            )
+        except Exception as exc:
+            error = exc
+        self.boundary_observation = {
+            "local_files_only": local_only, "ambient_file": ambient_file,
+            "exception_type": type(error).__name__ if error is not None else None,
+            "events": list(self.events), "model_assigned": hasattr(model, "model"),
+            "tokenizer_assigned": hasattr(model, "hf_tokenizer"),
+        }
+        self.assertIsInstance(error, ValueError, self.boundary_observation)
+        self.assertEqual(self.events, ["tokenizer.bytes"])
+        self.assertFalse(hasattr(model, "model"))
+        self.assertFalse(hasattr(model, "hf_tokenizer"))
+
+    def test_empty_buffer_local_with_ambient_file_refuses_before_allocation(self):
+        self._check_empty_buffer(True, True)
+
+    def test_empty_buffer_local_without_ambient_file_refuses_before_allocation(self):
+        self._check_empty_buffer(True, False)
+
+    def test_empty_buffer_nonlocal_with_ambient_file_refuses_before_allocation(self):
+        self._check_empty_buffer(False, True)
+
+    def test_empty_buffer_nonlocal_without_ambient_file_refuses_before_allocation(self):
