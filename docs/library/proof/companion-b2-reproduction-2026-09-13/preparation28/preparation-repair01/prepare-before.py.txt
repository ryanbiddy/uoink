def write(name, raw):
    target = out / name

    with target.open("xb") as stream:

def copy_original(name, target):
    raw = (old / name).read_bytes()

    write(target, raw)
    bindings.append({"source": str(old / name), "target": target, "bytes": len(raw), "sha256": sha(raw)})
    return raw
