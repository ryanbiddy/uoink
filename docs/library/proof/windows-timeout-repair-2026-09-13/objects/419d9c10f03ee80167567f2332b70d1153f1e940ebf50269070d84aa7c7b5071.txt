CASES = []


def case(name):
    def register(function):
        CASES.append((name, function))
        return function
    return register


def refused(kind, function, reason=None):
    try:
        function()
    except kind as error:
        if reason is not None:
            assert reason in str(error), (reason, str(error))
        return error
    raise AssertionError("Expected refusal: " + kind.__name__)


def channels():
    binding = P.GenerationBinding("1" * 64, "2" * 64, "3" * 64, "4" * 64, 123)
    return P.GenerationChannel(binding, b"x" * 32, "controller"), P.GenerationChannel(binding, b"x" * 32, "worker")


def signed_body(sender, body):
    header = P.HEADER.pack(P.MAGIC, sender._send_direction, sender._send_sequence, len(body))
    return header + body + hmac.digest(sender._key, sender._context + header + body, "sha256")

class FakeRegistry:
    def __init__(self, permit, factory_type):
        assert type(permit) is object
        self.permit, self.factory_type, self.active = permit, factory_type, True
        self.issued = []

    def assert_active(self):
        if not self.active:
            raise P.ProtocolRefusal("registry_revoked")

    def _issue_for_bootstrap(self, permit, factory, state_binding, recipe):
        assert permit is self.permit and type(factory) is self.factory_type
        self.assert_active()
        value = object()
        self.issued.append((factory, value))
        return value

    def revoke_generation(self, permit):
        assert permit is self.permit
        self.active = False


class FakeFactory:
    actions = {}
    events = []

    def __init__(self, fixed, tensor, runtime):
        self.runtime = runtime
        self.product = object()
        self._act("construct")

    def _act(self, name):
        self.events.append(name)
        action = self.actions.get(name)
        if isinstance(action, BaseException):
            raise action
        if action:
            action()

    def _bind_model_capability(self, capability):
        self.capability = capability
        self._act("bind")

    def build_strict_owned(self, state):
        self._act("build")
        return self.product

    def release_product(self, product):
        assert product is self.product
        self._act("release")


def boot(start=True):
    FakeFactory.actions, FakeFactory.events = {}, []
    a, b = channels()
    worker = P.WorkerBootstrap(b, FakeRegistry, FakeFactory)
    frame = a.encode("challenge", {"generation": a.binding.generation_hex, "namespace_sha256": a.binding.namespace_sha256})
    assert a.decode(worker.accept_challenge(frame))[0] == "ready"
    if start:
        worker.accept_begin(a.encode("begin", {"manifest_sha256": a.binding.manifest_sha256}))
    runtime = types.SimpleNamespace(_RUNTIME=worker.registry)
    return a, worker, runtime

class Scalar:
    def __init__(self, value=0):
        self.value = value


class MemoryBuffer:
    def __init__(self, value, size=None):
        self.raw = bytes(value) if type(value) is bytes else b"\0" * value
        if size is not None:
            assert size == len(self.raw)


class MemoryFFI:
    c_uint32 = Scalar
    c_int64 = Scalar
    create_string_buffer = MemoryBuffer

    @staticmethod
    def byref(value):
        return value

    @staticmethod
    def get_last_error():
        return 997


class MemoryAPI:
    def __init__(self, owner):
        self.ffi, self.owner = MemoryFFI(), owner
        self.actions = {}
        self.pending_counts = {}
        self.inbound = b""
        self.outbound = b""

    def structure(self, name):
        assert name == "OVERLAPPED"
        return types.SimpleNamespace(event=None)

    def checked(self, name, *args):
        result = self.call(name, *args)
        if not result:
            raise W.PipeRefusal("fake_api_failure_" + name)
        return result

    def call(self, name, *args):
        self.owner.events.append(name)
        action = self.actions.get(name)
        if isinstance(action, BaseException):
            raise action
        if callable(action):
            return action(*args)
        if action is not None:
            return action
        if name in ("CreateEventW", "CreateNamedPipeW", "CreateFileW"):
            self.owner.next_value += 1
            return self.owner.next_value
        if name == "ConnectNamedPipe":
            self.pending_counts[id(args[1])] = 0
            return 0
        if name == "WaitForSingleObject":
            return 0
        if name == "GetOverlappedResult":
            args[2].value = self.pending_counts[id(args[1])]
            return 1
        if name in ("GetHandleInformation", "SetHandleInformation", "CancelIoEx"):
            return 1
        if name == "GetFileType":
            return 3
        if name == "SetFilePointerEx":
            self.owner.positions[args[0]] = 0
            args[2].value = 0
            return 1
        if name == "WriteFile":
            handle, buffer, size, count, overlap = args
            self.outbound += buffer.raw[:size]
            self.pending_counts[id(overlap)] = size
            return 1
        if name == "ReadFile":
            handle, buffer, size, count, overlap = args
            if overlap is not None:
                value, self.inbound = self.inbound[:size], self.inbound[size:]
                self.pending_counts[id(overlap)] = len(value)
            else:
                at = self.owner.positions[handle]
                value = self.owner.contents[handle][at:at + size]
                self.owner.positions[handle] += len(value)
                count.value = len(value)
            buffer.raw = value + b"\0" * (size - len(value))
            return 1
        raise AssertionError("Unexpected fake API call: " + name)


class MemoryPrimitives:
    def __init__(self):
        self.handles, self.read_sets, self.workers, self.events = [], [], [], []
        self.contents, self.positions, self.identities = {}, {}, {}
        self.next_value = 100
        self.api = MemoryAPI(self)

    def _retain(self, value, kind):
        assert type(value) is int and value > 0
        handle = K.HandleRecord(value, kind)
        self.handles.append(handle)
        return handle

    def _owned(self, handle):
        assert any(item is handle for item in self.handles) and not handle.closed and not handle.unconfirmed
        return handle.value

    def _close(self, handle):
        self._owned(handle)
        self.events.append("close-" + handle.kind)
        handle.closed = True

    def identity(self, handle):
        self._owned(handle)
        return self.identities[handle.value]

    def create_suspended_worker(self, read_set, executable, arguments, cwd, environment, **options):
        assert options == {"inherited_control_handles": (self.control,), "include_control_handle_argument": True}
        worker = K.WorkerRecord(read_set=read_set, assigned=True, creation_time=123)
        self.workers.append(worker)
        read_set.worker_reserved = True
        return worker


def namespace(five=True):
    primitives = MemoryPrimitives()
    read_set = K.ReadSet()
    primitives.read_sets.append(read_set)
    assets = []
    for index, name in enumerate(sorted(N.FIVE if five else N.FOUR)):
        content = ("generated:" + name).encode()
        handle = primitives._retain(index + 1, "read-member")
        primitives.contents[handle.value] = content
        primitives.identities[handle.value] = K.FileIdentity("synthetic:" + name, 1, bytes([index]) * 16, len(content), 1, False)
        read_set.members.append(handle)
        read_set.identities.append(primitives.identities[handle.value])
        assets.append(N.BufferAsset(name, len(content), hashlib.sha256(content).hexdigest(), handle))
    return primitives, read_set, tuple(assets)


def materialize(rig):
    p, r, assets = rig
    return N.PinnedBufferNamespace(p, r, assets, sum(a.size for a in assets)).materialize()

def pipe_deadline():
    p = MemoryPrimitives()
    control = W.PrivatePipeController(p, lambda: 10.0)
    refused(W.PipeRefusal, lambda: control.create_pair(b"n" * 32, 100.0), "deadline")
    assert p.events == [] and control.pairs == []


@case("pipe_deadline_after_completed_chunk_quarantines_frame")
def pipe_partial_deadline():
    p, control, pair = pipe()
    times = iter((10.0, 10.0, 16.0))
    control.clock = lambda: next(times)
    refused(W.PipeRefusal, lambda: control.write_bytes(pair, b"x" * 17000, 15.0), "deadline")
    assert len(p.api.outbound) == W.IO_CHUNK and pair.operations == []
    assert pair.unconfirmed and not pair.active
    refused(W.PipeRefusal, lambda: control.write_bytes(pair, b"later", 15.0), "unavailable")


@case("pipe_overlapped_setup_failure_closes_recorded_event")
def pipe_partial_event():
    p, control, pair = pipe()
    error = KeyboardInterrupt("synthetic OVERLAPPED allocation interruption")
    def interrupted(name):
        raise error
    p.api.structure = interrupted
    assert refused(KeyboardInterrupt, lambda: control.write_bytes(pair, b"x", 15.0)) is error
    assert pair.unconfirmed and pair.operations == []
    assert all(h.closed for h in p.handles if h.kind == "ipc-event")
    assert not pair.server.closed and not pair.client.closed


@case("pipe_quarantine_stops_exact_worker_without_release_credit")
def pipe_forced_stop():
    p, control, pair = pipe()
    read_set = K.ReadSet()
    worker = K.WorkerRecord(read_set=read_set, assigned=True, creation_time=123)
    p.workers.append(worker)
    pair.worker, pair.read_set = worker, read_set
    stopped = []
    def stop(exact, timeout):
        assert exact is worker and timeout == 5000
        stopped.append(exact)
        return True
    p._stop_partial_worker = stop
    p.api.actions["WaitForSingleObject"] = 258
    refused(W.PipeRefusal, lambda: control.write_bytes(pair, b"generated", 15.0), "completion")
    assert stopped == [worker] and pair.forced_stop_attempted and pair.forced_process_stop_observed
    assert pair.unconfirmed and worker.unconfirmed and read_set.unconfirmed and not worker.quiescent
    assert len(pair.operations) == 1 and not pair.operations[0].event.closed
    refused(W.PipeRefusal, lambda: control.retire_endpoint(pair))
