"""Syntax comparisons of fixed own-source calls; never evaluate their targets."""
import ast
import copy

TARGETS = ('PyanNet', 'Specifications', 'load_state_dict', 'build',
           'requires_grad_', 'eval', 'VoiceActivitySegmentation', 'instantiate')


def factory_tree(text):
    begin = '----- BEGIN PROPOSED PYTHON (TEXT ONLY) -----'
    end = '----- END PROPOSED PYTHON -----'
    assert text.count(begin) == text.count(end) == 1
    return ast.parse(text.split(begin, 1)[1].split(end, 1)[0])


def function(tree, name):
    found = [node for node in ast.walk(tree) if isinstance(node, ast.FunctionDef) and node.name == name]
    assert len(found) == 1, name
    return found[0]


class Normalize(ast.NodeTransformer):
    def visit_Name(self, node):
        if node.id == 'plain_state':
            return ast.copy_location(ast.Name(id='state', ctx=node.ctx), node)
        return node

    def visit_Attribute(self, node):
        node = self.generic_visit(node)
        if isinstance(node.value, ast.Name) and node.value.id == 'factory':
            return ast.copy_location(ast.Name(id=node.attr, ctx=node.ctx), node)
        if isinstance(node.value, ast.Name) and node.value.id == 'owner' and node.attr in ('model', 'vad'):
            return ast.copy_location(ast.Name(id=node.attr, ctx=node.ctx), node)
        return node


def calls(tree):
    result = {}
    for node in ast.walk(tree):
        if not isinstance(node, ast.Call):
            continue
        name = node.func.attr if isinstance(node.func, ast.Attribute) else (
            node.func.id if isinstance(node.func, ast.Name) else None)
        if name in TARGETS:
            assert name not in result, 'Repeated fixed constructor/load call: ' + name
            normalized = Normalize().visit(copy.deepcopy(node))
            result[name] = ast.dump(normalized, include_attributes=False)
    assert set(result) == set(TARGETS)
    return result


def assert_fixed_calls(reference_text, proposed_text):
    reference = function(factory_tree(reference_text), 'build_fixed_candidate_after_release_gates')
    proposed = function(ast.parse(proposed_text), 'build_strict_owned')
    expected, actual = calls(reference), calls(proposed)
    assert expected == actual, [name for name in TARGETS if expected[name] != actual[name]]
    return tuple(TARGETS)


def helper_tree(reference_text):
    tree = factory_tree(reference_text)
    names = ('_fixed_shapes', '_check_plain_state')
    return ast.fix_missing_locations(ast.Module(body=[copy.deepcopy(function(tree, name)) for name in names],
                                                type_ignores=[]))


def constructor(tree, class_name):
    classes = [node for node in ast.walk(tree) if isinstance(node, ast.ClassDef) and node.name == class_name]
    assert len(classes) == 1
    initializers = [node for node in classes[0].body if isinstance(node, ast.FunctionDef) and node.name == '__init__']
    assert len(initializers) == 1
    return initializers[0]


def assert_owned_constructor(reference_text, harness_text):
    reference = copy.deepcopy(constructor(ast.parse(reference_text), 'VoiceActivitySegmentation'))
    actual = copy.deepcopy(constructor(ast.parse(harness_text), 'FakeVoiceActivitySegmentation'))
    for node in (reference, actual):
        for argument in (*node.args.posonlyargs, *node.args.args, *node.args.kwonlyargs):
            argument.annotation = None
        if node.args.vararg is not None:
            node.args.vararg.annotation = None
        if node.args.kwarg is not None:
            node.args.kwarg.annotation = None
    assert ast.dump(reference.args, include_attributes=False) == ast.dump(actual.args, include_attributes=False)
    assert [ast.dump(node, include_attributes=False) for node in reference.body] == [
        ast.dump(node, include_attributes=False) for node in actual.body]
